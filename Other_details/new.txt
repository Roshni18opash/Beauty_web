<script>
	document.addEventListener("DOMContentLoaded", function(){

		const totalItems = document.querySelectorAll("$_section .swiper-slide").length;
		const isMobile = window.innerWidth < 768;

		const enableSlider = isMobile
		? totalItems > {{= &m_slideperview}}
		: totalItems > {{= &slidepreview}};

		if(enableSlider){
			const swiperContainer = document.querySelector("$_section .trendingSwiper");
			if (swiperContainer) {
				swiperContainer.classList.add("slider-enabled");
			}
		}

		/* IF SLIDER ENABLED */
		if(enableSlider){
			new Swiper("$_section .trendingSwiper", {
				slidesPerView: {{= &slidepreview}},
				spaceBetween: 16,
				slidesPerGroup: 1,
				loop: {{= &slide_looping == 'y' ? "true" : "false"}},
				autoplay: {{= &auto_play == 'y' ? "{ delay: " + (&auto_play_speed * 1000) + ", disableOnInteraction: false }" : "false"}},

				on: {
					init: function () {
						setTimeout(() => {
							if(this.pagination && this.pagination.el) {
								const activeEl = this.pagination.el.querySelector('.swiper-pagination-bullet-active');
								if (activeEl) activeEl.classList.add('animate');
							}
						}, 10);
					},
					slideChange: function () {
						if(this.pagination && this.pagination.el) {
							const bullets = this.pagination.el.querySelectorAll('.swiper-pagination-bullet');
							bullets.forEach(b => b.classList.remove('animate'));
							setTimeout(() => {
								const activeEl = this.pagination.el.querySelector('.swiper-pagination-bullet-active');
								if (activeEl) activeEl.classList.add('animate');
							}, 10);
						}
					}
				},

				breakpoints: {
					0: {
						slidesPerView: {{= &m_slideperview}},
						spaceBetween: 10,
						centerInsufficientSlides: true,
						centeredSlides: false,

						pagination: {
							el: "$_section .swiper-pagination",
							clickable: true,
							enabled: {{= &m_indicator_option == "indicator" || &m_indicator_option == "both" ? "true" : "false"}},
							renderBullet: function (index, className) {
								return '<span class="' + className + '"><span class="progress"></span></span>';
							}
						},

						navigation: {
							nextEl: "$_section .custom-trending-next",
							prevEl: "$_section .custom-trending-prev",
							enabled: {{= &m_indicator_option == "arrow" || &m_indicator_option == "both" ? "true" : "false"}}
						}
					},
					768: {
						slidesPerView: {{= &slidepreview}},
						spaceBetween: {{= &spacebetween}},
						centeredSlides: false,

						pagination: {
							el: "$_section .swiper-pagination",
							clickable: true,
							enabled: {{= &indicator_options == "indicator" || &indicator_options == "both" ? "true" : "false"}},
							renderBullet: function (index, className) {
								return '<span class="' + className + '"><span class="progress"></span></span>';
							}
						},

						navigation: {
							nextEl: "$_section .custom-trending-next",
							prevEl: "$_section .custom-trending-prev",
							enabled: {{= &indicator_options == "arrow" || &indicator_options == "both" ? "true" : "false"}}
						},
						mousewheel: true,
						keyboard: true
					}
				}
			});
		} else {

			/* STATIC GRID */
			const swiperWrapper = document.querySelector("$_section .swiper-wrapper");
			if (swiperWrapper) {
				swiperWrapper.classList.add("static-grid");
			}

			const pagination = document.querySelector("$_section .swiper-pagination");
			const prev = document.querySelector("$_section .custom-trending-prev");
			const next = document.querySelector("$_section .custom-trending-next");

			if(pagination) pagination.style.display = "none";
			if(prev) prev.style.display = "none";
			if(next) next.style.display = "none";
		}
	});
</script>

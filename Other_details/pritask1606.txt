		let currentCardSlides = [...allCardSlides];

		let totalSlides = currentCardSlides.length;
		if (totalSlides === 0) {
			favoriteSection.style.display = 'none';
			return;
		}

		let currentIndex = 0;
		let isTransitioning = false;
		let autoplayTimer = null;

		const sliderConfig = {
			loop: {{= &slide_looping == 'y' ? 'true' : 'false'}},
			autoplay: {{= &auto_play == 'y' ? 'true' : 'false'}},
			autoplaySpeed: {{= &auto_play_speed ? &auto_play_speed : 5}} * 1000,
			transitionMs: 600,
			desktop: {
				arrows: {{= &indicator_options == 'arrow' || &indicator_options == 'both' ? 'true' : 'false'}},
				bullets: {{= &indicator_options == 'indicator' || &indicator_options == 'both' ? 'true' : 'false'}},
				counter: {{= &indicator_options == 'arrow' ? 'true' : 'false'}}
			},
			mobile: {
				arrows: {{= &mobile_indicator_options == 'arrow' || &mobile_indicator_options == 'both' ? 'true' : 'false'}},
				bullets: {{= &mobile_indicator_options == 'indicator' || &mobile_indicator_options == 'both' ? 'true' : 'false'}},
				counter: {{= &mobile_indicator_options == 'arrow' ? 'true' : 'false'}}
			}
		};

		const defaultBgColor = '#d8eae8';
		const bgColors = allTextSlides.map(slide => {
			const colorEl = slide.querySelector('.slide-bg-color');
			return colorEl && colorEl.dataset.color ? colorEl.dataset.color : defaultBgColor;
		});

		// Dynamic Tabs filtering by product category
		const categories = new Set();
		allCardSlides.forEach(slide => {
			const catEl = slide.querySelector('.product-category');
			if (catEl) {
				const cat = catEl.innerText.trim();
				if (cat) categories.add(cat);
			}
		});

		if (categories.size > 0) {
			const tabsWrapper = document.createElement('div');
			tabsWrapper.className = 'favorite-tabs-wrapper';
			
			const allTab = document.createElement('button');
			allTab.className = 'fav-tab-btn active';
			allTab.innerText = 'All';
			allTab.dataset.cat = 'All';
			tabsWrapper.appendChild(allTab);
			
			categories.forEach(cat => {
				const tab = document.createElement('button');
				tab.className = 'fav-tab-btn';
				tab.innerText = cat;
				tab.dataset.cat = cat;
				tabsWrapper.appendChild(tab);
			});
			
			// Insert tabs at the top of the section
			const wrapper = favoriteSection.querySelector('.favorite-wrapper');
			favoriteSection.insertBefore(tabsWrapper, wrapper);
			
			tabsWrapper.addEventListener('click', (e) => {
				if (e.target.classList.contains('fav-tab-btn')) {
					tabsWrapper.querySelectorAll('.fav-tab-btn').forEach(btn => btn.classList.remove('active'));
					e.target.classList.add('active');
					filterSlides(e.target.dataset.cat);
				}
			});
		}

		function filterSlides(category) {
			stopAutoplay();
			
			currentTextSlides = [];
			currentBgSlides = [];
			currentCardSlides = [];
			
			allCardSlides.forEach((cardSlide, i) => {
				const catEl = cardSlide.querySelector('.product-category');
				const cat = catEl ? catEl.innerText.trim() : '';
				
				const textSlide = allTextSlides[i];
				const bgSlide = allBgSlides[i];
				
				if (category === 'All' || cat === category) {
					cardSlide.style.display = '';
					textSlide.style.display = '';
					bgSlide.style.display = '';
					
					currentCardSlides.push(cardSlide);
					currentTextSlides.push(textSlide);
					currentBgSlides.push(bgSlide);
				} else {
					cardSlide.style.display = 'none';
					textSlide.style.display = 'none';
					bgSlide.style.display = 'none';
				}
			});
			
			totalSlides = currentCardSlides.length;
			currentIndex = 0;
			
			setSliderWidths();
			buildBullets();
			updateSlider(0);
			startAutoplay();
		}

		// Core Slider Logic
		function isMobile() { return window.innerWidth < 768; }
		
		function getActiveConfig() { return isMobile() ? sliderConfig.mobile : sliderConfig.desktop; }
		
		function getNavMode() {
			const cfg = getActiveConfig();
			if (cfg.arrows && cfg.bullets) return 'both';
			if (cfg.arrows && cfg.counter) return 'arrow';
			if (cfg.bullets) return 'indicator';
			return 'none';
		}

		function setSliderWidths() {
			if (totalSlides === 0) return;
			const slideWidth = 100 / totalSlides;
			if (textSliderTrack) textSliderTrack.style.width = `${totalSlides * 100}%`;
			currentTextSlides.forEach(slide => { slide.style.width = `${slideWidth}%`; });
			if (bgSliderTrack) bgSliderTrack.style.width = `${totalSlides * 100}%`;
			currentBgSlides.forEach(slide => { slide.style.width = `${slideWidth}%`; });
		}

		function buildBullets() {
			if (!sliderBullets) return;
			sliderBullets.innerHTML = '';
			for (let i = 0; i < totalSlides; i++) {
				const btn = document.createElement('button');
				btn.type = 'button';
				btn.className = 'slider-bullet';
				btn.setAttribute('aria-label', `Go to slide ${i + 1}`);
				btn.innerHTML = '<span class="bullet-progress"></span>';
				btn.addEventListener('click', () => goToSlide(i));
				sliderBullets.appendChild(btn);
			}
		}

		function updateBulletProgress() {
			if (!sliderBullets) return;
			const bullets = sliderBullets.querySelectorAll('.slider-bullet');

			bullets.forEach((dot, i) => {
				dot.classList.remove('active', 'animate');

				const progress = dot.querySelector('.bullet-progress');
				if (progress) {
					progress.style.transition = 'none';
					progress.style.width = '0%';
					void progress.offsetWidth;
				}

				if (i === currentIndex) {
					dot.classList.add('active');

					setTimeout(() => {
						if (progress) {
							progress.style.transition = '';
							progress.style.width = '';
						}
						dot.classList.add('animate');
					}, 10);
				}
			});
		}

		function updateNavUI() {
			if (sliderNavigation) sliderNavigation.setAttribute('data-mode', getNavMode());
			if (!sliderConfig.loop && totalSlides > 1) {
				if (prevBtn) prevBtn.disabled = currentIndex === 0;
				if (nextBtn) nextBtn.disabled = currentIndex === totalSlides - 1;
			} else {
				if (prevBtn) prevBtn.disabled = false;
				if (nextBtn) nextBtn.disabled = false;
			}
			
			if (totalSlides <= 1 && sliderNavigation) {
				sliderNavigation.style.setProperty('display', 'none', 'important');
			} else if (sliderNavigation) {
				sliderNavigation.style.removeProperty('display');
			}
		}

		function updateSlider(index) {
			if (totalSlides === 0) return;
			const slidePercent = 100 / totalSlides;
			const translateVal = `translateX(-${index * slidePercent}%)`;

			if (textSliderTrack) textSliderTrack.style.transform = translateVal;
			if (bgSliderTrack) bgSliderTrack.style.transform = translateVal;
			
			// Find original index to match the background color mapping
			const originalIndex = allCardSlides.indexOf(currentCardSlides[index]);
			if (leftPanel) leftPanel.style.backgroundColor = bgColors[originalIndex] || defaultBgColor;
			
			if (sliderCounter) sliderCounter.textContent = `${index + 1}/${totalSlides}`;

			currentTextSlides.forEach((s, i) => s.classList.toggle('active', i === index));
			currentBgSlides.forEach((s, i) => s.classList.toggle('active', i === index));
			currentCardSlides.forEach((s, i) => s.classList.toggle('card-slide-active', i === index));

			updateBulletProgress();
			updateNavUI();
		}

		function goToSlide(index) {
			if (isTransitioning || index === currentIndex) return;
			isTransitioning = true;
			currentIndex = index;
			updateSlider(currentIndex);
			setTimeout(() => { isTransitioning = false; }, sliderConfig.transitionMs);
			resetAutoplay();
		}

		function goNext() {
			if (isTransitioning || totalSlides <= 1) return;
			if (sliderConfig.loop) goToSlide((currentIndex + 1) % totalSlides);
			else if (currentIndex < totalSlides - 1) goToSlide(currentIndex + 1);
		}

		function goPrev() {
			if (isTransitioning || totalSlides <= 1) return;
			if (sliderConfig.loop) goToSlide((currentIndex - 1 + totalSlides) % totalSlides);
			else if (currentIndex > 0) goToSlide(currentIndex - 1);
		}

		function startAutoplay() {
			stopAutoplay();
			if (!sliderConfig.autoplay || totalSlides <= 1) return;
			autoplayTimer = setInterval(goNext, sliderConfig.autoplaySpeed);
		}

		function stopAutoplay() {
			if (autoplayTimer) {
				clearInterval(autoplayTimer);
				autoplayTimer = null;
			}
		}

		function resetAutoplay() {
			stopAutoplay();
			if (sliderConfig.autoplay) startAutoplay();
		}

		if (prevBtn) prevBtn.addEventListener('click', goPrev);
		if (nextBtn) nextBtn.addEventListener('click', goNext);

		if (favoriteSection) {
			favoriteSection.addEventListener('mouseenter', stopAutoplay);
			favoriteSection.addEventListener('mouseleave', () => {
				startAutoplay();
				updateBulletProgress();
			});
		}

		window.addEventListener('resize', updateNavUI);

		setSliderWidths();
		buildBullets();

		requestAnimationFrame(() => {
			requestAnimationFrame(() => {
				updateSlider(currentIndex);
				startAutoplay();
			});
		});
	});
});

<script>
	document.addEventListener("DOMContentLoaded", ()=> {
		let swiper = new Swiper("$_section-mySwiper", {

			slidesPerView: {{= &slidepreview}},
			spaceBetween: {{= &spacebetween}},
			slidesPerGroup: 1,

			loop: {{= &slide_looping == 'y' ? "true" : "false"}},

			autoplay: {{= &auto_play == 'y'
					? "{ delay: " + (&auto_play_speed * 1000) + ", disableOnInteraction: false }"
					: "false"
			}},

			on: {
				init: function () {
					setTimeout(() => {
						if(this.pagination && this.pagination.el) {
							const activeEl = this.pagination.el.querySelector('.swiper-pagination-bullet-active');
							if (activeEl) activeEl.classList.add('animate');
						}
					}, 10);
				},
				slideChange: function () {
					if(this.pagination && this.pagination.el) {
						const bullets = this.pagination.el.querySelectorAll('.swiper-pagination-bullet');
						bullets.forEach(b => b.classList.remove('animate'));
						setTimeout(() => {
							const activeEl = this.pagination.el.querySelector('.swiper-pagination-bullet-active');
							if (activeEl) activeEl.classList.add('animate');
						}, 10);
					}
				}
			},

			breakpoints: {

				0: {
					slidesPerView: {{= &item_to_show_mobile}},

					pagination: {
						el: ".$_s_class-swiper-pagination",
						clickable: true,
						enabled: {{= &m_indicator_options == "indicator" || &m_indicator_options == "both" ? "true" : "false"}},
						renderBullet: function (index, className) {
							return '<span class="' + className + '"><span class="progress"></span></span>';
						}
					},

					navigation: {
						nextEl: ".$_s_class-swiper-button-next",
						prevEl: ".$_s_class-swiper-button-prev",
						enabled: {{= &m_indicator_options == "arrow" || &m_indicator_options == "both" ? "true" : "false"}}
					}
				},

				768: {
					slidesPerView: {{= &slidepreview}},

					pagination: {
						el: ".$_s_class-swiper-pagination",
						enabled: {{= &indicator_options == "indicator" || &indicator_options == "both" ? "true" : "false"}},
						clickable: true,
						renderBullet: function (index, className) {
							return '<span class="' + className + '"><span class="progress"></span></span>';
						}
					},

					navigation: {
						nextEl: ".$_s_class-swiper-button-next",
						prevEl: ".$_s_class-swiper-button-prev",
						enabled: {{= &indicator_options == "arrow" || &indicator_options == "both" ? "true" : "false"}}
					},

					mousewheel: true,
					keyboard: true
				}
			}
		});

	})
	/* disabled click animation */


	document.addEventListener('click', function(e){

		const button = e.target.closest('.custom-nav-btn');

		if(!button) return;

		/* only disabled button */
		if(button.classList.contains('swiper-button-disabled')){

			/* avoid duplicate */
			if(button.querySelector('.blocked-icon')) return;

			/* create icon */
			const icon = document.createElement('span');

			icon.className = 'blocked-icon';

			icon.innerHTML = '⊘';

			/* add icon */
			button.appendChild(icon);

			/* remove after delay */
			setTimeout(() => {
				icon.remove();
			}, 700);
		}
	});

</script>

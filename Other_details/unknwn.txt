<script>
	var acc = document.getElementById("faqAccordion");

	acc.addEventListener("show.bs.collapse", function (e) {
		var btn = document.querySelector('[data-bs-target="#' + e.target.id + '"]');
		if (btn) btn.querySelector(".faq-icon").innerHTML = "&#8722;";
	});

	acc.addEventListener("hide.bs.collapse", function (e) {
		var btn = document.querySelector('[data-bs-target="#' + e.target.id + '"]');
		if (btn) btn.querySelector(".faq-icon").textContent = "+";
	});
</script>

		loop: {{= &slide_looping == 'y' ? "true" : "false"}},

		autoplay: {{= &auto_play == 'y'

		}},

		breakpoints: {

			0: {
				slidesPerView: {{= &item_to_show_mobile}},
				spaceBetween: 10,

				pagination: {
					el: ".swiper-pagination",
					clickable: true,
					enabled: {{= &mobile_indicator_options == "indicator" || &mobile_indicator_options == "both" ? "true" : "false"}}
				},

				navigation: {
					nextEl: ".swiper-button-next",
					prevEl: ".swiper-button-prev",
					enabled: {{= &mobile_indicator_options == "arrow" || &mobile_indicator_options == "both" ? "true" : "false"}}
				}
			},

					prevEl: ".swiper-button-prev",
					enabled: {{= &indicator_options == "arrow" || &indicator_options == "both" ? "true" : "false"}}
				},

				mousewheel: true,
				keyboard: true
			}
		}
	});

	/* disabled click animation */


	document.addEventListener('click', function(e){

		const button = e.target.closest('.custom-nav-btn');

		if(!button) return;

		/* only disabled button */
		if(button.classList.contains('swiper-button-disabled')){

			/* avoid duplicate */
			if(button.querySelector('.blocked-icon')) return;

			/* create icon */
			const icon = document.createElement('span');

			icon.className = 'blocked-icon';

			icon.innerHTML = '⊘';

			/* add icon */
			button.appendChild(icon);

			/* remove after delay */
			setTimeout(() => {
				icon.remove();
			}, 700);
		}
	});






</script>

<script>
	document.addEventListener('DOMContentLoaded', () => {

		// ══════════════════════════════════════════════════════
		// DOM ELEMENTS & GLOBAL STATE
		// ══════════════════════════════════════════════════════
		const showMoreBtn   = document.getElementById('show-more-btn');
		const cardWrappers  = Array.from(document.querySelectorAll('.review-card-wrapper'));
		let isExpanded      = false;

		if (!showMoreBtn || cardWrappers.length === 0) return;

		// ══════════════════════════════════════════════════════
		// Kitne cards initially dikhane hain — screen size ke hisaab se
		// Mobile (< 768px) → 3 cards | Tablet/Desktop → 6 cards
		// ══════════════════════════════════════════════════════
		function getInitialCount() {
			return window.innerWidth < 768 ? 3 : 6;
		}

		// ══════════════════════════════════════════════════════
		// Page load par sahi cards dikhao, baaki hide karo
		// ══════════════════════════════════════════════════════
		function applyInitialState() {
			const initialCount = getInitialCount();

			cardWrappers.forEach((wrapper, idx) => {
				const card = wrapper.querySelector('.review-card');

				if (idx < initialCount) {
					wrapper.classList.remove('d-none');
					if (card) {
						card.classList.remove('card-faded', 'fade-in');
						// Last visible card par faded "peek" effect (agar aur cards hidden hain)
						if (idx === initialCount - 1 && cardWrappers.length > initialCount) {
							card.classList.add('card-faded');
						}
					}
				} else {
					wrapper.classList.add('d-none');
					if (card) card.classList.remove('fade-in', 'card-faded');
				}
			});

			// Agar total cards hi initial count se kam/equal hain to button hide karo
			showMoreBtn.style.display = cardWrappers.length <= initialCount ? 'none' : 'inline-block';
			showMoreBtn.textContent = 'Show More';
			isExpanded = false;
		}

		// ══════════════════════════════════════════════════════
		// Show More / Show Less toggle
		// ══════════════════════════════════════════════════════
		function toggleReviews() {
			const initialCount = getInitialCount();

			if (!isExpanded) {
				// ── SHOW MORE: baaki saare cards dikhao ──
				cardWrappers.forEach((wrapper, idx) => {
					const card = wrapper.querySelector('.review-card');
					wrapper.classList.remove('d-none');
					if (card) card.classList.remove('card-faded');

					if (idx >= initialCount && card) {
						card.style.animationDelay = ((idx - initialCount) * 0.08) + 's';
						card.classList.add('fade-in');
					}
				});

				showMoreBtn.textContent = 'Show Less';
				isExpanded = true;

			} else {
				// ── SHOW LESS: wapas initial count par le aao ──
				cardWrappers.forEach((wrapper, idx) => {
					const card = wrapper.querySelector('.review-card');
					if (card) card.classList.remove('fade-in');

					if (idx >= initialCount) {
						wrapper.classList.add('d-none');
					} else if (idx === initialCount - 1 && card) {
						card.classList.add('card-faded');
					}
				});

				document.getElementById('reviews-section').scrollIntoView({
					behavior: 'smooth',
					block: 'start'
				});

				showMoreBtn.textContent = 'Show More';
				isExpanded = false;
			}
		}

		// ══════════════════════════════════════════════════════
		// INITIALIZE
		// ══════════════════════════════════════════════════════
		applyInitialState();
		showMoreBtn.addEventListener('click', toggleReviews);

		// Screen resize hone par (mobile↔desktop switch) — sirf collapsed state me re-check karo
		let resizeTimer;
		window.addEventListener('resize', () => {
			clearTimeout(resizeTimer);
			resizeTimer = setTimeout(() => {
				if (!isExpanded) applyInitialState();
			}, 200);
		});
	});



</script>

<script>
  document.addEventListener("DOMContentLoaded", () => {
    // ══════════════════════════════════════════════════════
    // DOM ELEMENTS & GLOBAL STATE
    // ══════════════════════════════════════════════════════
    const showMoreBtn = document.getElementById("show-more-btn");
    const cardWrappers = Array.from(
      document.querySelectorAll(".review-card-wrapper"),
    );
    let isExpanded = false;

    if (!showMoreBtn || cardWrappers.length === 0) return;

    // ══════════════════════════════════════════════════════
    // Kitne cards initially dikhane hain — screen size ke hisaab se
    // Mobile (< 768px) → 3 cards | Tablet/Desktop → 6 cards
    // ══════════════════════════════════════════════════════
    function getInitialCount() {
      return window.innerWidth < 768 ? 3 : 6;
    }

    // ══════════════════════════════════════════════════════
    // Page load par sahi cards dikhao, baaki hide karo
    // ══════════════════════════════════════════════════════
    function applyInitialState() {
      const initialCount = getInitialCount();

      cardWrappers.forEach((wrapper, idx) => {
        const card = wrapper.querySelector(".review-card");

        if (idx < initialCount) {
          wrapper.classList.remove("d-none");
          if (card) {
            card.classList.remove("card-faded", "fade-in");
            // Last visible card par faded "peek" effect (agar aur cards hidden hain)
            if (
              idx === initialCount - 1 &&
              cardWrappers.length > initialCount
            ) {
              card.classList.add("card-faded");
            }
          }
        } else {
          wrapper.classList.add("d-none");
          if (card) card.classList.remove("fade-in", "card-faded");
        }
      });

      // Agar total cards hi initial count se kam/equal hain to button hide karo
      showMoreBtn.style.display =
        cardWrappers.length <= initialCount ? "none" : "inline-block";
      showMoreBtn.textContent = "Show More";
      isExpanded = false;
    }

    // ══════════════════════════════════════════════════════
    // Show More / Show Less toggle
    // ══════════════════════════════════════════════════════
    function toggleReviews() {
      const initialCount = getInitialCount();

      if (!isExpanded) {
        // ── SHOW MORE: baaki saare cards dikhao ──
        cardWrappers.forEach((wrapper, idx) => {
          const card = wrapper.querySelector(".review-card");
          wrapper.classList.remove("d-none");
          if (card) card.classList.remove("card-faded");

          if (idx >= initialCount && card) {
            card.style.animationDelay = (idx - initialCount) * 0.08 + "s";
            card.classList.add("fade-in");
          }
        });

        showMoreBtn.textContent = "Show Less";
        isExpanded = true;
      } else {
        // ── SHOW LESS: wapas initial count par le aao ──
        cardWrappers.forEach((wrapper, idx) => {
          const card = wrapper.querySelector(".review-card");
          if (card) card.classList.remove("fade-in");

          if (idx >= initialCount) {
            wrapper.classList.add("d-none");
          } else if (idx === initialCount - 1 && card) {
            card.classList.add("card-faded");
          }
        });

        document.getElementById("reviews-section").scrollIntoView({
          behavior: "smooth",
          block: "start",
        });

        showMoreBtn.textContent = "Show More";
        isExpanded = false;
      }
    }

    // ══════════════════════════════════════════════════════
    // INITIALIZE
    // ══════════════════════════════════════════════════════
    applyInitialState();
    showMoreBtn.addEventListener("click", toggleReviews);

    // Screen resize hone par (mobile↔desktop switch) — sirf collapsed state me re-check karo
    let resizeTimer;
    window.addEventListener("resize", () => {
      clearTimeout(resizeTimer);
      resizeTimer = setTimeout(() => {
        if (!isExpanded) applyInitialState();
      }, 200);
    });
  });
</script>
